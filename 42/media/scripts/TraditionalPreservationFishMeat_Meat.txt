    /* Rendering takes no station: a pot, two pieces of raw fat, then any stove or fire. */
    craftRecipe PutPorkFatInPot
    {
        timedAction = MixingBowl,
        time = 50,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [Base.Pot;Base.PotForged] mode:destroy flags[InheritCondition;IsEmpty] mappers[potType],
            item 2 [TraditionalPreservationFishMeat.PorkFat] flags[InheritFoodAge;IsUncookedFoodItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:potType,
        }
        itemMapper potType
        {
            TraditionalPreservationFishMeat.PotOfPorkFat = Base.Pot,
            TraditionalPreservationFishMeat.PotForgedPorkFat = Base.PotForged,
        }
    }

    /* A cooked pot's fat has melted: pouring it off gives the pot back holding 0.5 L of Melted Lard or Melted
       Tallow (onPourOffLard, onPourOffTallow fill it), for Strain Lard or Strain Tallow, a bigger container or
       the Hearth's copper. IsCookedFoodItem lets burnt food through, so canPourOff refuses a burnt pot. */
    craftRecipe PourOffLard
    {
        timedAction = MixingBowl,
        time = 60,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationFishMeat.canPourOff,
        OnCreate = TraditionalPreservationFishMeat.onPourOffLard,
        inputs
        {
            item 1 [TraditionalPreservationFishMeat.PotOfPorkFat;TraditionalPreservationFishMeat.PotForgedPorkFat] mappers[potType] flags[InheritCondition;IsCookedFoodItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:potType,
        }
        itemMapper potType
        {
            Base.Pot = TraditionalPreservationFishMeat.PotOfPorkFat,
            Base.PotForged = TraditionalPreservationFishMeat.PotForgedPorkFat,
        }
    }

    /* Melted fat from the Hearth's copper (TraditionalPreservationFishMeat_fluids.txt), in whatever it was ladled
       into: half a litre strains to a block and its cracklings, as a pot's two pieces do. It goes through a cloth or
       a Strainer (a Cheese Cloth, Dish Towel, clean Rag or Sheet), which is kept, as the container is. */
    craftRecipe StrainLard
    {
        timedAction = MixingBowl,
        time = 60,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        xpAward = Cooking:5,
        inputs
        {
            item 1 [Base.CheeseCloth;Base.DishCloth;Base.RippedSheets;Base.Sheet;Base.Strainer] mode:keep,
            item 1 [*],
            -fluid 0.5 [MeltedLard],
        }
        outputs
        {
            item 1 Base.Lard,
            item 1 TraditionalPreservationFishMeat.Cracklings,
        }
    }

    {
        timedAction = MixingBowl,
        time = 80,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        OnTest = TraditionalPreservationFishMeat.canScrape,
        inputs
        {
            item 1 [TraditionalPreservationFishMeat.PotOfPorkFat;TraditionalPreservationFishMeat.PotForgedPorkFat;TraditionalPreservationFishMeat.PotOfSuet;TraditionalPreservationFishMeat.PotForgedSuet] mappers[potType] flags[InheritCondition;AllowRottenItem;ItemCount] mode:destroy,
        }
        outputs
        {
            item 1 mapper:potType,
        }
        itemMapper potType
        {
            Base.Pot = TraditionalPreservationFishMeat.PotOfPorkFat,
            Base.Pot = TraditionalPreservationFishMeat.PotOfSuet,
            Base.PotForged = TraditionalPreservationFishMeat.PotForgedPorkFat,
