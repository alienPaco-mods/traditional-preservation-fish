       which is how onCreateCured finds it. Don't use IsWholeFoodItem: it needs the script's exact
       hunger, and fishing and these stages all change that. */
    craftRecipe PackFishInSalt
    {
        timedAction = SliceFish,
        time = 100,
        Tags = InHandCraft;CanBeDoneFromFloor;Cooking,
        category = Cooking,
        xpAward = Cooking:3,
        OnTest = TraditionalPreservationFishMeat.canPack,
        OnCreate = TraditionalPreservationFishMeat.onCreatePacked,

            item 1 tags[base:sharpknife;base:meatcleaver] mode:keep flags[IsNotDull;SharpnessCheck],
